🎯 QUICK START - 10 MINUTES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1️⃣  Generate SSH Key (locally)
    ssh-keygen -t rsa -b 4096 -f deploy_key -N ""

2️⃣  Add to Server
    ssh-copy-id -i deploy_key.pub user@servidor

3️⃣  Configure GitHub Secrets (5 required)
    GitHub → Settings → Secrets

    SSH_PRIVATE_KEY = [from deploy_key]
    SSH_HOST = servidor.com
    SSH_USER = deploy
    SSH_PROJECT_PATH = /var/www/mail-alert
    ALERT_SECRET = your_secret_key

4️⃣  Push to Main
    git add backend/
    git commit -m "feature: your change"
    git push origin main

    ✅ Workflow runs automatically!
